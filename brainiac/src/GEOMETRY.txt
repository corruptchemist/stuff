// GEOMETRY 0.5 -- Brainiac's geometry program for the TI-84 Plus CE.
//
// Built by  python -m brainiac.build  into GEOMETRY.8xp. Brainiac's menu
// runs it with prgmGEOMETRY; it also runs on its own from [prgm].
// Same source conventions as BRAINIAC.txt (see build.py): // notes,
// @labels, #sub/#call/#return. Its #call stack lives in list GS.
//
// Every solver first draws a reference figure with letters on it, so you
// can see which side or angle each prompt means, then asks for the values,
// then draws the figure again with the numbers and shows the working. Drawing is done in pixels: the window is
// set to 0..264 by 0..164 (the CE graph area), so Line( and Text( can be
// given pixel positions; Text( rows count from the top, so row=164-y.
//
// Variables: M = mode (for AGAIN), R = next home-screen row, θ = return
// address (#return). The helpers only ever touch G I J N P Q (and Z Y, the
// sizes given to ring and oval), so a solver can keep its inputs in any
// other letter. tx draws Str0 at pixel (P,Q).
// Lists: PX, PY the figure's points (pixels, y up); Z1-Z4 spare points;
// GW the window and angle mode to put back when leaving.

ClrHome
// remember the calculator's own settings, then use degrees and pixels
{abs(sin(90)-1)<ᴇ⁻9,Xmin,Xmax,Ymin,Ymax,Xscl,Yscl}→ʟGW
// the axes, grid and which Y= functions are on, to put back at the end
StoreGDB 9
0→dim(ʟGS)
Degree
// a split screen (Horiz, G-T) would make the graph too short for the
// labels; Fix or Sci display would change how numbers read
Full
Normal

264→Xmax
0→Ymin
164→Ymax

// ======================================================== helpers

// X -> Str0, rounded to 3 places (fa: angles, 2 places).
#sub fm
round(X,3)→X
toString(X)→Str0
#return
#sub fa
toString(round(X,2))→Str0
#return

// Show Str0 from home-screen row R down, turning the page when full.
// Changes J.
#sub line
0→J
For(I,1,length(Str0))
J+1+(sub(Str0,I,1)="√(")→J
End
iPart((J-1)/26)+1→J
If R+J>10:Then
Output(10,1,"ENTER: MORE")
Pause
ClrHome
1→R
End
Output(R,1,Str0)
R+J→R
#return

// A reference picture is up: wait, then clear for the prompts.
#sub ref
Text(150,2,"ENTER, THEN TYPE THE VALUES")
Pause 
ClrHome
#return

// The drawing is done: wait, then start the home-screen report.
#sub shown
Text(150,2,"ENTER FOR THE WORKING")
Pause
ClrHome
1→R
#return

// The report is done: wait for ENTER.
#sub done
Output(10,1,"ENTER: DONE")
Pause
#return

// Fit the points in PX, PY (any units, y up) into the picture box, keeping
// their shape; they become pixel positions. Changes G J N P Q.
#sub fit
min(ʟPX)→P
max(ʟPX)→Q
min(ʟPY)→G
max(ʟPY)→N
If Q=P:P+1→Q
If N=G:G+1→N
min(196/(Q-P),104/(N-G))→J
34+J(ʟPX-P)+(196-J(Q-P))/2→ʟPX
30+J(ʟPY-G)+(104-J(N-G))/2→ʟPY
#return

// Join the points in PX, PY in order, and back to the first. Changes I J N.
#sub poly
dim(ʟPX)→N
For(I,1,N)
I+1-N(I=N)→J
Line(ʟPX(I),ʟPY(I),ʟPX(J),ʟPY(J))
End
#return

// Str0 centred at pixel (P,Q). Text( wants whole numbers (ERR:DOMAIN
// otherwise), and the text is kept on the screen, at about 8 pixels a character (the CE graph font).
#sub tx
Text(iPart(max(0,min(150,164-Q-4))),iPart(max(0,min(264-8length(Str0),P-4length(Str0)))),Str0)
#return

// Str0 just inside corner I of the figure (an angle label). Changes G J P Q.
#sub vlab
sum(ʟPX)/dim(ʟPX)→G
sum(ʟPY)/dim(ʟPY)→J
ʟPX(I)+.3(G-ʟPX(I))→P
ʟPY(I)+.3(J-ʟPY(I))→Q
#call tx
#return

// Str0 just outside the side from corner I to the next (a side label).
// Changes G J N P Q.
#sub slab
dim(ʟPX)→N
I+1-N(I=N)→J
sum(ʟPX)/N→G
(ʟPX(I)+ʟPX(J))/2→P
P+.16(P-G)→P
sum(ʟPY)/N→G
(ʟPY(I)+ʟPY(J))/2→Q
Q+.16(Q-G)→Q
#call tx
#return

// A small square in corner I, marking a right angle. Changes G J N P Q.
#sub rmark
dim(ʟPX)→N
I+1-N(I=N)→J
ʟPX(J)-ʟPX(I)→P
ʟPY(J)-ʟPY(I)→Q
7/√(P²+Q²)→G
PG→P
QG→Q
I-1+N(I=1)→J
ʟPX(J)-ʟPX(I)→G
ʟPY(J)-ʟPY(I)→N
7/√(G²+N²)→J
GJ→G
NJ→N
Line(ʟPX(I)+P,ʟPY(I)+Q,ʟPX(I)+P+G,ʟPY(I)+Q+N)
Line(ʟPX(I)+G,ʟPY(I)+N,ʟPX(I)+P+G,ʟPY(I)+Q+N)
#return

// A triangle with sides A, B, C: corner 1 is where B and C meet (so it has
// angle A), corner 2 where C and A meet, corner 3 where A and B meet. Side
// C runs along the bottom. The corners go into PX, PY, fitted.
#sub tri3
{0,C,(B²+C²-A²)/(2C)}→ʟPX
{0,0,√(max(0,B²-ʟPX(3)²))}→ʟPY
#call fit
#return

// Draw the triangle in PX, PY with angle labels Str1 Str2 Str3 at corners
// 1 2 3 and side labels Str4 Str5 Str6 on sides 1-2, 2-3, 3-1.
#sub tridraw
ClrDraw
#call poly
1→I
Str1→Str0
#call vlab
2→I
Str2→Str0
#call vlab
3→I
Str3→Str0
#call vlab
1→I
Str4→Str0
#call slab
2→I
Str5→Str0
#call slab
3→I
Str6→Str0
#call slab
#return

// A circle of radius Z pixels at (P,Q), drawn with Line( so that it is
// round whatever the window. Changes I.
#sub ring
For(I,0,35)
Line(P+Zcos(10I),Q+Zsin(10I),P+Zcos(10I+10),Q+Zsin(10I+10))
End
#return

// An ellipse at (P,Q), half-widths Z across and Y up. Changes I.
#sub oval
For(I,0,35)
Line(P+Zcos(10I),Q+Ysin(10I),P+Zcos(10I+10),Q+Ysin(10I+10))
End
#return
